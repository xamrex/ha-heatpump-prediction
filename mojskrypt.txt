"""
PyScript do obliczania średniej temperatury dzisiejszej (aktualna + forecast).
Łączy rzeczywiste pomiary z sensor.aktualna_pogoda z prognozą pogody.
 
Sensor: sensor.srednia_temp_today_actualandforcast
Atrybuty:
  - srednia_aktualna: średnia z rzeczywistych pomiarów (od 00:00 do teraz)
  - srednia_forecast: średnia z prognozy (od teraz do końca dnia)
  - wartosci_aktualne: lista wszystkich pomiarów
  - wartosci_forecast: lista wszystkich wartości z prognozy
  - liczba_pomiarow_aktualnych: ilość pomiarów historycznych
  - liczba_godzin_forecast: ilość godzin z prognozy
 
Uruchamiane: Co godzinę o pełnej godzinie
"""
 
from datetime import datetime, timedelta
 
def get_actual_temp_yesterday_avg(sensor_id="sensor.heatpump2_outdoor_ambient_temperature"):
    """
    Pobiera średnią temperaturę z podanego sensora z wczoraj (00:00–24:00).
 
    Args:
        sensor_id (str): nazwa sensora HA
 
    Returns:
        tuple: (średnia_temperatura, liczba_pomiarów, lista_wartości) lub (None, 0, [])
    """
    try:
        import sqlite3
        db_path = "/config/home-assistant_v2.db"
 
        conn = sqlite3.connect(db_path)
        cursor = conn.cursor()
 
        # Północ dziś
        today_midnight = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
 
        # Północ wczoraj
        yesterday_midnight = today_midnight - timedelta(days=1)
 
        ts_start = yesterday_midnight.timestamp()
        ts_end = today_midnight.timestamp()
 
        query = """
        SELECT CAST(s.state AS REAL)
        FROM states s
        JOIN states_meta sm ON s.metadata_id = sm.metadata_id
        WHERE sm.entity_id = ?
          AND s.last_updated_ts >= ?
          AND s.last_updated_ts < ?
          AND s.state NOT IN ('unknown', 'unavailable', 'none', '')
          AND CAST(s.state AS REAL) BETWEEN -50 AND 50
        ORDER BY s.last_updated_ts
        """
 
        cursor.execute(query, (sensor_id, ts_start, ts_end))
        results = cursor.fetchall()
        conn.close()
 
        if results:
            values = [round(float(row[0]), 1) for row in results]
            avg_temp = round(sum(values) / len(values), 2)
            count = len(values)
 
            log.info(
                f"Średnia temperatura z wczoraj ({sensor_id}): "
                f"{avg_temp}°C (z {count} pomiarów)"
            )
 
            return avg_temp, count, values
        else:
            log.warning(f"Brak pomiarów temperatury z wczoraj ({sensor_id})")
            return None, 0, []
 
    except Exception as e:
        log.error(f"Błąd pobierania temperatury z wczoraj ({sensor_id}): {e}")
        import traceback
        log.error(traceback.format_exc())
        return None, 0, []

def get_forecast_temp_tomorrow_avg(weather_entity="weather.forecast_dom"):
    """
    Pobiera średnią temperaturę z prognozy na jutro (00:00–24:00).
 
    Args:
        weather_entity (str): encja weather
 
    Returns:
        tuple: (średnia_temperatura, liczba_godzin, lista_wartości) lub (None, 0, [])
    """
    try:
        # Pobierz prognozę
        response = service.call(
            "weather",
            "get_forecasts",
            entity_id=weather_entity,
            type="hourly",
            return_response=True,
            blocking=True
        )
 
        if not response or weather_entity not in response:
            log.error(f"Nie udało się pobrać prognozy ({weather_entity})")
            return None, 0, []
 
        forecast_data = response[weather_entity]["forecast"]
 
        # Zakres jutra
        today_midnight = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
        tomorrow_start = today_midnight + timedelta(days=1)
        tomorrow_end = tomorrow_start + timedelta(days=1)
 
        tomorrow_temps = []
 
        for forecast in forecast_data:
            forecast_time = datetime.fromisoformat(
                forecast["datetime"]
            ).replace(tzinfo=None)
 
            if tomorrow_start <= forecast_time < tomorrow_end:
                temp = round(float(forecast["temperature"]), 1)
                tomorrow_temps.append(temp)
 
        if tomorrow_temps:
            avg_temp = round(sum(tomorrow_temps) / len(tomorrow_temps), 2)
 
            log.info(
                f"Średnia temperatura jutro ({weather_entity}): "
                f"{avg_temp}°C (z {len(tomorrow_temps)} godzin)"
            )
            log.info(f"Wartości: {tomorrow_temps}")
 
            return avg_temp, len(tomorrow_temps), tomorrow_temps
        else:
            log.warning(f"Brak prognoz temperatur na jutro ({weather_entity})")
            return None, 0, []
 
    except Exception as e:
        log.error(f"Błąd pobierania prognozy ({weather_entity}): {e}")
        import traceback
        log.error(traceback.format_exc())
        return None, 0, []

@time_trigger("cron(1 * * * *)")
def update_srednia_temp_today():
    """
    Aktualizuje sensor średniej temperatury dzisiejszej co godzinę.
    Łączy rzeczywiste pomiary z prognozą.
    """
    log.info("=" * 60)
    log.info("AKTUALIZACJA ŚREDNIEJ TEMPERATURY DZISIAJ (AKTUALNA + FORECAST)")
    log.info("=" * 60)
 
    try:
        # Pobierz średnią z rzeczywistych pomiarów
        srednia_aktualna, liczba_pomiarow, wartosci_aktualne = get_actual_temp_today_avg()
 
        # Pobierz średnią z prognozy
        srednia_forecast, liczba_godzin_forecast, wartosci_forecast = get_forecast_temp_today_avg()
 
        # Jeśli nie ma żadnych danych, ustaw sensor jako unavailable
        if srednia_aktualna is None and srednia_forecast is None:
            log.error("Brak danych - ani pomiarów, ani prognozy")
            state.set("sensor.srednia_temp_today_actualandforcast", "unavailable",
                     new_attributes={
                         "friendly_name": "Średnia temperatura dzisiaj (aktualna+forecast)",
                         "unit_of_measurement": "°C",
                         "device_class": "temperature",
                         "error": "Brak danych",
                         "last_update": datetime.now().isoformat()
                     })
            return
 
        # Oblicz średnią ważoną
        # Jeśli mamy oba źródła, liczymy średnią ważoną liczba pomiarów/godzin
        # Jeśli mamy tylko jedno źródło, używamy go
        if srednia_aktualna is not None and srednia_forecast is not None:
            # Średnia ważona: (suma_temp_aktualnych + suma_temp_forecast) / (liczba_aktualnych + liczba_forecast)
            total_weight = liczba_pomiarow + liczba_godzin_forecast
            srednia_laczna = round(
                (srednia_aktualna * liczba_pomiarow + srednia_forecast * liczba_godzin_forecast) / total_weight,
                1
            )
        elif srednia_aktualna is not None:
            srednia_laczna = srednia_aktualna
            srednia_forecast = None
            liczba_godzin_forecast = 0
            wartosci_forecast = []
        else:
            srednia_laczna = srednia_forecast
            srednia_aktualna = None
            liczba_pomiarow = 0
            wartosci_aktualne = []
 
        log.info(f"Wyniki:")
        log.info(f"  Średnia aktualna: {srednia_aktualna}°C (z {liczba_pomiarow} pomiarów)")
        log.info(f"  Średnia forecast: {srednia_forecast}°C (z {liczba_godzin_forecast} godzin)")
        log.info(f"  Średnia łączna: {srednia_laczna}°C")
 
        # Zapisz sensor
        state.set("sensor.srednia_temp_today_actualandforcast", float(srednia_laczna),
            new_attributes = {
                "unit_of_measurement": "°C",
                "device_class": "temperature",
                "state_class": "measurement",
                "friendly_name": "Średnia temperatura dzisiaj (aktualna+forecast)",
                "srednia_aktualna": srednia_aktualna,
                "srednia_forecast": srednia_forecast,
                "wartosci_aktualne": wartosci_aktualne,
                "wartosci_forecast": wartosci_forecast,
                "liczba_pomiarow_aktualnych": liczba_pomiarow,
                "liczba_godzin_forecast": liczba_godzin_forecast,
                "last_update": datetime.now().isoformat(),
                "source_sensor": "sensor.aktualna_pogoda",
                "source_weather": "weather.forecast_dom"
            })
        log.info(f"✅ Sensor zaktualizowany: {srednia_laczna}°C")
 
    except Exception as e:
        log.error(f"Błąd podczas aktualizacji sensora: {e}")
        import traceback
        log.error(traceback.format_exc())

                     new_attributes={
                         "friendly_name": "Średnia wilgotność dzisiaj (aktualna+forecast)",
                         "unit_of_measurement": "%",
                         "device_class": "humidity",
                         "error": "Brak danych",
                         "last_update": datetime.now().isoformat()
                     })
            return
 
        # Oblicz średnią ważoną
        if srednia_aktualna is not None and srednia_forecast is not None:
            total_weight = liczba_pomiarow + liczba_godzin_forecast
            srednia_laczna = round(
                (srednia_aktualna * liczba_pomiarow + srednia_forecast * liczba_godzin_forecast) / total_weight,
                1
            )
        elif srednia_aktualna is not None:
            srednia_laczna = srednia_aktualna
            srednia_forecast = None
            liczba_godzin_forecast = 0
            wartosci_forecast = []
        else:
            srednia_laczna = srednia_forecast
            srednia_aktualna = None
            liczba_pomiarow = 0
            wartosci_aktualne = []
 
        log.info(f"Wyniki:")
        log.info(f"  Średnia aktualna: {srednia_aktualna}% (z {liczba_pomiarow} pomiarów)")
        log.info(f"  Średnia forecast: {srednia_forecast}% (z {liczba_godzin_forecast} godzin)")
        log.info(f"  Średnia łączna: {srednia_laczna}%")
 
 
        # Zapisz sensor
        state.set("sensor.srednia_humidity_today_actualandforcast", float(srednia_laczna),
            new_attributes = {
                "unit_of_measurement": "%",
                "device_class": "humidity",
                "state_class": "measurement",
                "friendly_name": "Średnia wilgotność dzisiaj (aktualna+forecast)",
                "srednia_aktualna": srednia_aktualna,
                "srednia_forecast": srednia_forecast,
                "wartosci_aktualne": wartosci_aktualne,
                "wartosci_forecast": wartosci_forecast,
                "liczba_pomiarow_aktualnych": liczba_pomiarow,
                "liczba_godzin_forecast": liczba_godzin_forecast,
                "last_update": datetime.now().isoformat(),
                "source_sensor": "sensor.aktualnapogodawilgotnosc",
                "source_weather": "weather.forecast_dom"
            })
        log.info(f"✅ Sensor wilgotności zaktualizowany: {srednia_laczna}%")
 
    except Exception as e:
        log.error(f"Błąd podczas aktualizacji sensora wilgotności: {e}")
        import traceback
        log.error(traceback.format_exc())

def check_source_sensors_ready():
    """
    Sprawdza czy sensory źródłowe są gotowe i dostępne.
 
    Returns:
        bool: True jeśli sensory są gotowe, False w przeciwnym razie
    """
    try:
        # Sprawdź sensor temperatury
        temp_state = state.get("sensor.aktualna_pogoda")
        if not temp_state or temp_state in ["unknown", "unavailable", None]:
            log.warning("Sensor aktualna_pogoda nie jest jeszcze dostępny")
            return False
 
        # Sprawdź sensor wilgotności
        humidity_state = state.get("sensor.aktualnapogodawilgotnosc")
        if not humidity_state or humidity_state in ["unknown", "unavailable", None]:
            log.warning("Sensor aktualnapogodawilgotnosc nie jest jeszcze dostępny")
            return False
 
        # Sprawdź weather entity
        weather_state = state.get("weather.forecast_dom")
        if not weather_state or weather_state in ["unknown", "unavailable", None]:
            log.warning("Weather forecast_dom nie jest jeszcze dostępny")
            return False
 
        log.info("✅ Wszystkie sensory źródłowe są dostępne")
        return True
 
    except Exception as e:
        log.error(f"Błąd sprawdzania sensorów źródłowych: {e}")
        return False
 
 
@time_trigger("startup")
def init_sensors_on_reload():
    """
    Inicjalizuje sensory przy przeładowaniu PyScript.
    @time_trigger("startup") uruchamia się raz przy załadowaniu modułu.
    Używa retry logic aby poczekać aż źródłowe sensory będą dostępne.
    """
    log.info("=" * 60)
    log.info("PYSCRIPT STARTUP - Inicjalizacja sensorów pogodowych")
    log.info("=" * 60)
 
    # Czekaj 10 sekund żeby PyScript, baza danych i integracje się w pełni załadowały
    log.info("Czekam 10 sekund na inicjalizację Home Assistant...")
    task.sleep(10)
 
    # Retry logic - 3 próby z 5s opóźnieniem między próbami
    max_attempts = 3
    for attempt in range(1, max_attempts + 1):
        log.info(f"Próba {attempt}/{max_attempts} - sprawdzam dostępność sensorów źródłowych...")
 
        if check_source_sensors_ready():
            # Sensory są dostępne, możemy aktualizować
            log.info("Uruchamiam aktualizację sensora temperatury...")
            update_srednia_temp_today()
 
            log.info("Uruchamiam aktualizację sensora temperatury wczoraj...")
            update_srednia_temp_yesterday()
 
            log.info("Uruchamiam aktualizację sensora temperatury jutrzejszej...")
            update_srednia_temp_tomorrow()
 
            log.info("Uruchamiam aktualizację sensora wilgotności...")
            update_srednia_humidity_today()
            update_srednia_humidity_tomorrow()
 
            log.info("✅ Inicjalizacja sensorów po przeładowaniu zakończona")
            return
 
        # Jeśli to nie była ostatnia próba, czekaj 5 sekund przed następną
        if attempt < max_attempts:
            log.warning(f"Sensory nie są gotowe, czekam 5 sekund przed próbą {attempt + 1}...")
            task.sleep(5)
 
    # Jeśli po 3 próbach sensory nadal nie są dostępne
    log.error("❌ Nie udało się zainicjalizować sensorów - źródłowe sensory niedostępne")
    log.info("Sensory zaktualizują się automatycznie przy następnym cron trigger (XX:01)")
